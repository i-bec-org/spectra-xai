spectraxai.dataset

The module spectraxai.dataset includes classes and operations on datasets.

This module defines the class Dataset which ought to be used to handle all necessary dataset transformations including scaling, spectral pre-processing, and splitting. The classes defined in spectraxai.models and spectraxai.explain generally expect as inputs a Dataset object to operate on.

273class Dataset:
274    """A general class to manage the dataset (i.e. input X and output Y).
275
276    Use this class to pass your 2D spectral matrix and 1D or 2D output
277    properties. Supports methods for pre-processing X, scaling X and Y,
278    splitting the dataset, and more.
279
280    """
281
282    n_samples: int
283    """Number of samples in the dataset."""
284
285    n_features: int
286    """Number of features in the dataset's input."""
287
288    n_outputs: int
289    """Number of outputs in the dataset's output."""
290
291    X: np.ndarray
292    """The spectral data of size (`n_samples`, `n_features`)"""
293
294    Y: np.ndarray
295    """The output properties of size (`n_samples`, `n_outputs`)"""
296
297    X_names: list[str | int]
298    """The names of the input features of size (`n_features`)"""
299
300    Y_names: list[str]
301    """The names of the output properties of size (`n_outputs`)"""
302
303    def __init__(
304        self,
305        X: np.ndarray | pandas.DataFrame,
306        Y: np.ndarray | pandas.DataFrame | pandas.Series,
307        X_names: Sequence[str | int] | np.ndarray | None = None,
308        Y_names: Sequence[str] | np.ndarray | None = None,
309        *,
310        wavelengths: Sequence[float] | np.ndarray | None = None,
311    ) -> None:
312        """Create a new Dataset from input and output arrays.
313
314        Parameters
315        ----------
316        X: `numpy.ndarray` or `pandas.DataFrame`
317            A 2D matrix of the spectra of size (`n_samples`, `n_features`)
318
319        Y: `numpy.ndarray` or `pandas.DataFrame` or `pandas.Series`
320            A 1D vector (`n_samples`,) or 2D matrix (`n_samples`, `n_outputs`)
321            of the output property(ies). If 1D it will be implicitly converted
322            to 2D.
323
324        X_names: `list[str]`, optional
325            A list of length `n_features` containing the names of the input
326            features. If missing, these will be autogenerated as X1, X2, etc.
327
328        Y_names: `list[str]`, optional
329            A list of length `n_outputs` containing the names of the output
330            properties. When using a single output property pass a list of
331            size 1. If missing, these will be autogenerated as Y1, Y2, etc.
332
333        wavelengths: sequence of `float`, optional
334            The wavelength of each input feature, of length `n_features`. If
335            missing, `wavelengths` is derived from `X_names` when these are numeric.
336
337        """
338        if isinstance(X, pandas.DataFrame):
339            X = X.to_numpy()
340        if isinstance(Y, pandas.DataFrame | pandas.Series):
341            Y = Y.to_numpy()
342        if X.shape[0] != Y.shape[0]:
343            raise SpectraXAIValueError("X and Y don't have the same number of rows!")
344        if X.ndim != 2:
345            raise SpectraXAIValueError("X should have exactly two dimensions")
346        self.X = X
347        self.Y = Y if Y.ndim > 1 else Y.reshape(-1, 1)
348        check_X_y(self.X, self.Y, multi_output=True)
349        self.n_samples = self.X.shape[0]
350        self.n_features = self.X.shape[1]
351        self.n_outputs = self.Y.shape[1]
352        x_names = [] if X_names is None else list(X_names)
353        y_names = [] if Y_names is None else list(Y_names)
354        if x_names and len(x_names) != self.n_features:
355            raise SpectraXAIValueError("X_names should have the same n_features as X")
356        if y_names and len(y_names) != self.n_outputs:
357            raise SpectraXAIValueError("Y_names should have the same n_outputs as Y")
358        self.X_names = x_names or ["X" + str(i + 1) for i in range(self.n_features)]
359        self.Y_names = y_names or ["Y" + str(i + 1) for i in range(self.n_outputs)]
360        self._wavelengths = None
361        if wavelengths is not None:
362            wvs = np.asarray(wavelengths, dtype=float).ravel()
363            if wvs.size != self.n_features or not np.isfinite(wvs).all():
364                raise SpectraXAIValueError(
365                    "wavelengths should contain n_features finite values"
366                )
367            self._wavelengths = wvs
368
369    @property
370    def wavelengths(self) -> np.ndarray | None:
371        """The wavelength of each input feature, or None if unknown.
372
373        Either the `wavelengths` passed to the constructor, or `X_names` converted to
374        floats when they are all numeric (e.g. ``[350, 370, ...]`` or
375        ``["350", "370", ...]``).
376
377        """
378        if self._wavelengths is not None:
379            return self._wavelengths.copy()
380        try:
381            wvs = np.asarray(self.X_names, dtype=float)
382        except (TypeError, ValueError):
383            return None
384        return wvs if np.isfinite(wvs).all() else None
385
386    def _like(self, X: np.ndarray, Y: np.ndarray) -> "Dataset":
387        """Return a new Dataset with the same names and wavelengths."""
388        return Dataset(X, Y, self.X_names, self.Y_names, wavelengths=self._wavelengths)

390    def to_dataframe(self) -> pandas.DataFrame:
391        """Return the dataset as a single `pandas.DataFrame`.
392
393        Columns are the input features (named by `X_names`, as strings) followed by
394        the outputs (named by `Y_names`), e.g. to pass to tools such as pycaret.
395
396        """
397        columns = [str(name) for name in self.X_names] + list(self.Y_names)
398        if len(set(columns)) != len(columns):
399            raise SpectraXAIValueError("X_names and Y_names should be unique")
400        return pandas.DataFrame(np.hstack((self.X, self.Y)), columns=columns)

408    def train_test_split(
409        self,
410        split: DatasetSplit,
411        opt: int | float,
412        *,
413        random_state: int | None = None,
414        n_bins: int = 10,
415    ) -> DataSplit:
416        """Split dataset with passed split method to train and test.
417
418        Parameters
419        ----------
420        split: `DatasetSplit`
421            The method used to split the dataset
422
423        opt: `Number`
424            A float number (between 0 and 1) indicating the percentage of the
425            training dataset for Random and Kennard–Stone split. A natural number
426            for Cross Validation and Stratified split.
427
428        random_state: `int`, optional
429            Seed for the RANDOM, CROSS_VALIDATION and STRATIFIED splits. Defaults to
430            None (non-reproducible, as in earlier versions).
431
432        n_bins: `int`, optional
433            For the STRATIFIED split of a continuous output: the number of quantile
434            bins of the (first) output used as strata. Class-like outputs (binary
435            or multiclass) are stratified on their labels directly. Defaults to 10.
436
437        Returns
438        -------
439        `DataSplit`
440            The idx_trn, idx_tst tuple. Use e.g. X[idx_trn], Y[idx_trn] to get the
441            training dataset.
442
443        """
444        indices = np.arange(self.X.shape[0])
445        if split == DatasetSplit.RANDOM or split == DatasetSplit.KENNARD_STONE:
446            msg = (
447                "opt param should be a float in the (0, 1) range for "
448                "RANDOM and KENNARD_STONE splits"
449            )
450            if not isinstance(opt, float):
451                raise TypeError(msg)
452            if opt <= 0 or opt >= 1:
453                raise SpectraXAIValueError(msg)
454        elif split == DatasetSplit.CROSS_VALIDATION or split == DatasetSplit.STRATIFIED:
455            msg = (
456                "opt param should be a positive int greater than 1 for "
457                "CROSS_VALIDATION and STRATIFIED splits"
458            )
459            if not isinstance(opt, int):
460                raise TypeError(msg)
461            if opt <= 1:
462                raise SpectraXAIValueError(msg)
463        if split == DatasetSplit.RANDOM:
464            idx_trn, idx_tst = train_test_split(
465                indices, train_size=opt, random_state=random_state
466            )
467            return np.asarray(idx_trn), np.asarray(idx_tst)
468        elif split == DatasetSplit.KENNARD_STONE:
469            _, _, idx_trn, idx_tst = kennard_stone.train_test_split(
470                self.X, indices, test_size=(1 - opt)
471            )
472            return idx_trn, idx_tst
473        elif split == DatasetSplit.CLHS:
474            raise NotImplementedError("clhs not implemented yet")
475        elif split == DatasetSplit.CROSS_VALIDATION:
476            assert isinstance(opt, int)
477            kf = KFold(opt, shuffle=True, random_state=random_state)
478            idx_trn, idx_tst = [], []
479            for trn_index, tst_index in kf.split(self.X):
480                idx_trn.append(np.array(trn_index))
481                idx_tst.append(np.array(tst_index))
482            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)
483        elif split == DatasetSplit.STRATIFIED:
484            assert isinstance(opt, int)
485            if random_state is None:
486                skf = StratifiedKFold(n_splits=opt)
487            else:
488                skf = StratifiedKFold(
489                    n_splits=opt, shuffle=True, random_state=random_state
490                )
491            idx_trn, idx_tst = [], []
492            for trn_index, tst_index in skf.split(self.X, self._strata(n_bins)):
493                idx_trn.append(np.array(trn_index))
494                idx_tst.append(np.array(tst_index))
495            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)
496
497    def _strata(self, n_bins: int) -> np.ndarray:
498        """Labels to stratify on: class labels, or quantile bins of output 1."""
499        y = self.Y[:, 0] if self.n_outputs > 1 else self.Y.ravel()
500        if type_of_target(y) in ("binary", "multiclass"):
501            return y
502        if n_bins < 2:
503            raise SpectraXAIValueError("n_bins should be at least 2")
504        return np.asarray(
505            pandas.qcut(y, q=n_bins, labels=False, duplicates="drop"), dtype=int
506        )

508    def train_test_split_explicit(
509        self, trn: np.ndarray | None = None, tst: np.ndarray | None = None
510    ) -> DataSplit:
511        """Split dataset to train and test from pre-selected by the user indices.
512
513        This is useful in cases where one has the set containing the training
514        indices of a dataset and wants to get the complement of the training set to
515        obtain the test set.
516
517        Parameters
518        ----------
519        trn: `np.ndarray`, optional
520            Contains the indices of the training samples
521
522        tst: `np.ndarray`, optional
523            Contains the indices of the testing samples
524
525        Returns
526        -------
527        `DataSplit`
528            The idx_trn, idx_tst tuple
529
530        """
531        trn = np.array([]) if trn is None else trn
532        tst = np.array([]) if tst is None else tst
533        if tst.size == 0 and trn.size == 0:
534            raise SpectraXAIValueError("You need to specify either tst or trn indices")
535        if tst.size > 0 and trn.size > 0:
536            raise SpectraXAIValueError("You cannot specify both trn and tst")
537        train_indices: np.ndarray | list[np.ndarray] = trn
538        test_indices: np.ndarray | list[np.ndarray] = tst
539        if tst.size > 0 and isinstance(tst[0], np.ndarray):
540            train_indices = [
541                self.train_test_split_explicit(tst=fold)[0] for fold in tst
542            ]
543        elif tst.size > 0:
544            if not np.logical_and(tst >= 0, tst < self.n_samples).all():
545                raise SpectraXAIValueError("Passed indices contain out of bound values")
546            train_indices = np.array(
547                list(set(range(0, self.n_samples)).difference(set(tst)))
548            )
549        elif trn.size > 0 and isinstance(trn[0], np.ndarray):
550            test_indices = [self.train_test_split_explicit(trn=fold)[1] for fold in trn]
551        else:
552            if not np.logical_and(trn >= 0, trn < self.n_samples).all():
553                raise SpectraXAIValueError("Passed indices contain out of bound values")
554            test_indices = np.array(
555                list(set(range(0, self.n_samples)).difference(set(trn)))
556            )
557        trn, tst = np.array(train_indices), np.array(test_indices)
558        if isinstance(trn[0], np.ndarray) and trn[0].dtype == "O":
559            trn = trn.astype(np.int64)
560        if isinstance(tst[0], np.ndarray) and tst[0].dtype == "O":
561            tst = tst.astype(np.int64)
562        return trn, tst
