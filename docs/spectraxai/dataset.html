spectraxai.dataset

The module spectraxai.dataset includes classes and operations on datasets.

This module defines the class Dataset which ought to be used to handle all necessary dataset transformations including scaling, spectral pre-processing, and splitting. The classes defined in spectraxai.models and spectraxai.explain generally expect as inputs a Dataset object to operate on.

155            return X, scaler.get_params(), Scale.__get_scale_attributes(method, scaler)
156        elif len(set_params) != 0 and len(set_attributes) != 0:
157            return X
158        elif len(set_attributes) != 0:

266def _clhs_objective(
267    data: np.ndarray, strata: np.ndarray, corr: np.ndarray, idx: np.ndarray, n: int
268) -> float:
269    """cLHS objective: strata coverage error plus correlation error."""
270    counts = np.stack(
271        [np.bincount(strata[idx, j], minlength=n) for j in range(data.shape[1])],
272        axis=1,
273    )
274    objective = float(np.abs(counts - 1).sum())
275    if data.shape[1] > 1:
276        objective += float(np.abs(np.corrcoef(data[idx], rowvar=False) - corr).sum())
277    return objective

280def _clhs(
281    data: np.ndarray, n: int, iterations: int, rng: np.random.Generator
282) -> np.ndarray:
283    """Conditioned Latin hypercube sampling (Minasny & McBratney, 2006).
284
285    Selects ``n`` rows of ``data`` (samples x continuous variables) so that each
286    variable's ``n`` quantile strata hold about one selected sample each and the
287    correlations between variables are preserved, using simulated annealing with
288    random swaps between selected and unselected samples.
289
290    Returns
291    -------
292    `np.ndarray`
293        The sorted indices of the selected samples.
294
295    """
296    n_samples, n_vars = data.shape
297    edges = np.quantile(data, np.linspace(0, 1, n + 1), axis=0)
298    strata = np.stack(
299        [
300            np.clip(np.searchsorted(edges[1:-1, j], data[:, j], side="right"), 0, n - 1)
301            for j in range(n_vars)
302        ],
303        axis=1,
304    )
305    corr = np.corrcoef(data, rowvar=False) if n_vars > 1 else np.ones((1, 1))
306    order = rng.permutation(n_samples)
307    selected, unselected = order[:n].copy(), order[n:].copy()
308    objective = _clhs_objective(data, strata, corr, selected, n)
309    best_objective, best = objective, selected.copy()
310    temperature, cooling = 1.0, 0.95
311    for iteration in range(iterations):
312        i, j = rng.integers(n), rng.integers(n_samples - n)
313        selected[i], unselected[j] = unselected[j], selected[i]
314        candidate = _clhs_objective(data, strata, corr, selected, n)
315        delta = candidate - objective
316        if delta <= 0 or rng.random() < np.exp(-delta / temperature):
317            objective = candidate
318            if objective < best_objective:
319                best_objective, best = objective, selected.copy()
320        else:  # undo the swap
321            selected[i], unselected[j] = unselected[j], selected[i]
322        if (iteration + 1) % 10 == 0:
323            temperature *= cooling
324    return np.sort(best)

339def _normalize_unit(unit: str | None) -> str | None:
340    if unit is None:

358class Dataset:
359    """A general class to manage the dataset (i.e. input X and output Y).
360
361    Use this class to pass your 2D spectral matrix and 1D or 2D output
362    properties. Supports methods for pre-processing X, scaling X and Y,
363    splitting the dataset, and more.
364
365    """
366
367    n_samples: int
368    """Number of samples in the dataset."""
369
370    n_features: int
371    """Number of features in the dataset's input."""
372
373    n_outputs: int
374    """Number of outputs in the dataset's output."""
375
376    X: np.ndarray
377    """The spectral data of size (`n_samples`, `n_features`)"""
378
379    Y: np.ndarray
380    """The output properties of size (`n_samples`, `n_outputs`)"""
381
382    X_names: list[str | int | float]
383    """The names of the input features of size (`n_features`)"""
384
385    Y_names: list[str]
386    """The names of the output properties of size (`n_outputs`)"""
387
388    spectral_unit: str | None
389    """The unit of `wavelengths`: ``"nm"``, ``"um"``, ``"cm-1"`` or None."""
390
391    def __init__(
392        self,
393        X: np.ndarray | pandas.DataFrame,
394        Y: np.ndarray | pandas.DataFrame | pandas.Series,
395        X_names: Sequence[str | int | float] | np.ndarray | None = None,
396        Y_names: Sequence[str] | np.ndarray | None = None,
397        *,
398        wavelengths: Sequence[float] | np.ndarray | None = None,
399        spectral_unit: str | None = None,
400    ) -> None:
401        """Create a new Dataset from input and output arrays.
402
403        Parameters
404        ----------
405        X: `numpy.ndarray` or `pandas.DataFrame`
406            A 2D matrix of the spectra of size (`n_samples`, `n_features`)
407
408        Y: `numpy.ndarray` or `pandas.DataFrame` or `pandas.Series`
409            A 1D vector (`n_samples`,) or 2D matrix (`n_samples`, `n_outputs`)
410            of the output property(ies). If 1D it will be implicitly converted
411            to 2D.
412
413        X_names: `list[str]`, optional
414            A list of length `n_features` containing the names of the input
415            features. If missing, these will be autogenerated as X1, X2, etc.
416
417        Y_names: `list[str]`, optional
418            A list of length `n_outputs` containing the names of the output
419            properties. When using a single output property pass a list of
420            size 1. If missing, these will be autogenerated as Y1, Y2, etc.
421
422        wavelengths: sequence of `float`, optional
423            The wavelength of each input feature, of length `n_features`. If
424            missing, `wavelengths` is derived from `X_names` when these are numeric.
425            For mid-infrared data these are wavenumbers (see ``spectral_unit``).
426
427        spectral_unit: `str`, optional
428            The unit of `wavelengths`: ``"nm"``, ``"um"`` or ``"cm-1"`` (also
429            accepted: ``"µm"``, ``"cm^-1"``, ``"1/cm"``). With ``"cm-1"``
430            (wavenumbers, e.g. MIR spectra) plots label the axis "Wavenumber
431            (cm⁻¹)" and draw it from high to low, as is customary, whatever the
432            order of the columns. Defaults to None (unknown; axes are labelled
433            "Wavelength").
434
435        """
436        if isinstance(X, pandas.DataFrame):
437            X = X.to_numpy()
438        if isinstance(Y, pandas.DataFrame | pandas.Series):
439            Y = Y.to_numpy()
440        if X.shape[0] != Y.shape[0]:
441            raise SpectraXAIValueError("X and Y don't have the same number of rows!")
442        if X.ndim != 2:
443            raise SpectraXAIValueError("X should have exactly two dimensions")
444        self.X = X
445        self.Y = Y if Y.ndim > 1 else Y.reshape(-1, 1)
446        check_X_y(self.X, self.Y, multi_output=True)
447        self.n_samples = self.X.shape[0]
448        self.n_features = self.X.shape[1]
449        self.n_outputs = self.Y.shape[1]
450        x_names = [] if X_names is None else list(X_names)
451        y_names = [] if Y_names is None else list(Y_names)
452        if x_names and len(x_names) != self.n_features:
453            raise SpectraXAIValueError("X_names should have the same n_features as X")
454        if y_names and len(y_names) != self.n_outputs:
455            raise SpectraXAIValueError("Y_names should have the same n_outputs as Y")
456        self.X_names = x_names or ["X" + str(i + 1) for i in range(self.n_features)]
457        self.Y_names = y_names or ["Y" + str(i + 1) for i in range(self.n_outputs)]
458        self.spectral_unit = _normalize_unit(spectral_unit)
459        self._wavelengths = None
460        if wavelengths is not None:
461            wvs = np.asarray(wavelengths, dtype=float).ravel()
462            if wvs.size != self.n_features or not np.isfinite(wvs).all():
463                raise SpectraXAIValueError(
464                    "wavelengths should contain n_features finite values"
465                )
466            self._wavelengths = wvs
467
468    @property
469    def wavelengths(self) -> np.ndarray | None:

471
472        Either the `wavelengths` passed to the constructor, or `X_names` converted to

512        """Return the dataset as a single `pandas.DataFrame`.

529    def train_test_split(
530        self,
531        split: DatasetSplit,
532        opt: int | float,
533        *,
534        random_state: int | None = None,
535        n_bins: int = 10,
536        clhs_iterations: int = 5000,
537        clhs_components: int = 5,
538    ) -> DataSplit:
539        """Split dataset with passed split method to train and test.
540
541        Parameters
542        ----------
543        split: `DatasetSplit`
544            The method used to split the dataset
545
546        opt: `Number`
547            A float number (between 0 and 1) indicating the percentage of the
548            training dataset for Random, Kennard–Stone and cLHS split. A natural
549            number for Cross Validation and Stratified split.
550
551        random_state: `int`, optional
552            Seed for the RANDOM, CLHS, CROSS_VALIDATION and STRATIFIED splits.
553            Defaults to None (non-reproducible, as in earlier versions).
554
555        n_bins: `int`, optional
556            For the STRATIFIED split of a continuous output: the number of quantile
557            bins of the (first) output used as strata. Class-like outputs (binary
558            or multiclass) are stratified on their labels directly. Defaults to 10.
559
560        clhs_iterations: `int`, optional
561            For the CLHS split: the number of simulated-annealing iterations.
562            Defaults to 5000.
563
564        clhs_components: `int`, optional
565            For the CLHS split: the number of principal components of the
566            (standardized) spectra that the sample should cover. Defaults to 5.
567
568        Returns
569        -------
570        `DataSplit`
571            The idx_trn, idx_tst tuple. Use e.g. X[idx_trn], Y[idx_trn] to get the
572            training dataset.
573
574        """
575        indices = np.arange(self.X.shape[0])
576        if split in (
577            DatasetSplit.RANDOM,
578            DatasetSplit.KENNARD_STONE,
579            DatasetSplit.CLHS,
580        ):
581            msg = (
582                "opt param should be a float in the (0, 1) range for "
583                "RANDOM, KENNARD_STONE and CLHS splits"
584            )
585            if not isinstance(opt, float):
586                raise TypeError(msg)
587            if opt <= 0 or opt >= 1:
588                raise SpectraXAIValueError(msg)
589        elif split == DatasetSplit.CROSS_VALIDATION or split == DatasetSplit.STRATIFIED:
590            msg = (
591                "opt param should be a positive int greater than 1 for "
592                "CROSS_VALIDATION and STRATIFIED splits"
593            )
594            if not isinstance(opt, int):
595                raise TypeError(msg)
596            if opt <= 1:
597                raise SpectraXAIValueError(msg)
598        if split == DatasetSplit.RANDOM:
599            idx_trn, idx_tst = train_test_split(
600                indices, train_size=opt, random_state=random_state
601            )
602            return np.asarray(idx_trn), np.asarray(idx_tst)
603        elif split == DatasetSplit.KENNARD_STONE:
604            _, _, idx_trn, idx_tst = kennard_stone.train_test_split(
605                self.X, indices, test_size=(1 - opt)
606            )
607            return idx_trn, idx_tst
608        elif split == DatasetSplit.CLHS:
609            n_select = int(round(opt * self.n_samples))
610            if not 1 <= n_select < self.n_samples:
611                raise SpectraXAIValueError("opt leaves an empty training or test set")
612            n_components = max(
613                1, min(clhs_components, self.n_features, self.n_samples - 1)
614            )
615            scores = PCA(
616                n_components=n_components, random_state=random_state
617            ).fit_transform(StandardScaler().fit_transform(self.X))
618            idx_trn = _clhs(
619                scores, n_select, clhs_iterations, np.random.default_rng(random_state)
620            )
621            idx_tst = np.setdiff1d(indices, idx_trn)
622            return idx_trn, idx_tst

531        self,
532        split: DatasetSplit,
533        opt: int | float,
534        *,
535        random_state: int | None = None,
536        n_bins: int = 10,
537        clhs_iterations: int = 5000,
538        clhs_components: int = 5,
539    ) -> DataSplit:
540        """Split dataset with passed split method to train and test.
541
542        Parameters
543        ----------
544        split: `DatasetSplit`
545            The method used to split the dataset
546
547        opt: `Number`
548            A float number (between 0 and 1) indicating the percentage of the
549            training dataset for Random, Kennard–Stone and cLHS split. A natural
550            number for Cross Validation and Stratified split.
551
552        random_state: `int`, optional
553            Seed for the RANDOM, CLHS, CROSS_VALIDATION and STRATIFIED splits.
554            Defaults to None (non-reproducible, as in earlier versions).
555
556        n_bins: `int`, optional
557            For the STRATIFIED split of a continuous output: the number of quantile
558            bins of the (first) output used as strata. Class-like outputs (binary
559            or multiclass) are stratified on their labels directly. Defaults to 10.
560
561        clhs_iterations: `int`, optional
562            For the CLHS split: the number of simulated-annealing iterations.
563            Defaults to 5000.
564
565        clhs_components: `int`, optional
566            For the CLHS split: the number of principal components of the
567            (standardized) spectra that the sample should cover. Defaults to 5.
568
569        Returns
570        -------
571        `DataSplit`
572            The idx_trn, idx_tst tuple. Use e.g. X[idx_trn], Y[idx_trn] to get the
573            training dataset.
574
575        """
576        indices = np.arange(self.X.shape[0])
577        if split in (
578            DatasetSplit.RANDOM,
579            DatasetSplit.KENNARD_STONE,
580            DatasetSplit.CLHS,
581        ):
582            msg = (
583                "opt param should be a float in the (0, 1) range for "
584                "RANDOM, KENNARD_STONE and CLHS splits"
585            )
586            if not isinstance(opt, float):
587                raise TypeError(msg)
588            if opt <= 0 or opt >= 1:
589                raise SpectraXAIValueError(msg)
590        elif split == DatasetSplit.CROSS_VALIDATION or split == DatasetSplit.STRATIFIED:
591            msg = (
592                "opt param should be a positive int greater than 1 for "
593                "CROSS_VALIDATION and STRATIFIED splits"
594            )
595            if not isinstance(opt, int):
596                raise TypeError(msg)
597            if opt <= 1:
598                raise SpectraXAIValueError(msg)
599        if split == DatasetSplit.RANDOM:
600            idx_trn, idx_tst = train_test_split(
601                indices, train_size=opt, random_state=random_state
602            )
603            return np.asarray(idx_trn), np.asarray(idx_tst)
604        elif split == DatasetSplit.KENNARD_STONE:
605            _, _, idx_trn, idx_tst = kennard_stone.train_test_split(
606                self.X, indices, test_size=(1 - opt)
607            )
608            return idx_trn, idx_tst
609        elif split == DatasetSplit.CLHS:
610            n_select = int(round(opt * self.n_samples))
611            if not 1 <= n_select < self.n_samples:
612                raise SpectraXAIValueError("opt leaves an empty training or test set")
613            n_components = max(
614                1, min(clhs_components, self.n_features, self.n_samples - 1)
615            )
616            scores = PCA(
617                n_components=n_components, random_state=random_state
618            ).fit_transform(StandardScaler().fit_transform(self.X))
619            idx_trn = _clhs(
620                scores, n_select, clhs_iterations, np.random.default_rng(random_state)
621            )
622            idx_tst = np.setdiff1d(indices, idx_trn)
623            return idx_trn, idx_tst
624        elif split == DatasetSplit.CROSS_VALIDATION:
625            assert isinstance(opt, int)
626            kf = KFold(opt, shuffle=True, random_state=random_state)
627            idx_trn, idx_tst = [], []
628            for trn_index, tst_index in kf.split(self.X):
629                idx_trn.append(np.array(trn_index))
630                idx_tst.append(np.array(tst_index))
631            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)
632        elif split == DatasetSplit.STRATIFIED:
633            assert isinstance(opt, int)
634            if random_state is None:
635                skf = StratifiedKFold(n_splits=opt)
636            else:
637                skf = StratifiedKFold(
638                    n_splits=opt, shuffle=True, random_state=random_state
639                )
640            idx_trn, idx_tst = [], []
641            for trn_index, tst_index in skf.split(self.X, self._strata(n_bins)):
642                idx_trn.append(np.array(trn_index))
643                idx_tst.append(np.array(tst_index))
644            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)

647        """Labels to stratify on: class labels, or quantile bins of output 1."""

580            DatasetSplit.CLHS,
581        ):
582            msg = (

584                "RANDOM, KENNARD_STONE and CLHS splits"
585            )
586            if not isinstance(opt, float):
587                raise TypeError(msg)
588            if opt <= 0 or opt >= 1:
589                raise SpectraXAIValueError(msg)
590        elif split == DatasetSplit.CROSS_VALIDATION or split == DatasetSplit.STRATIFIED:
591            msg = (
592                "opt param should be a positive int greater than 1 for "
593                "CROSS_VALIDATION and STRATIFIED splits"
594            )
595            if not isinstance(opt, int):
596                raise TypeError(msg)
597            if opt <= 1:
598                raise SpectraXAIValueError(msg)
599        if split == DatasetSplit.RANDOM:
600            idx_trn, idx_tst = train_test_split(
601                indices, train_size=opt, random_state=random_state
602            )
603            return np.asarray(idx_trn), np.asarray(idx_tst)
604        elif split == DatasetSplit.KENNARD_STONE:
605            _, _, idx_trn, idx_tst = kennard_stone.train_test_split(
606                self.X, indices, test_size=(1 - opt)
607            )
608            return idx_trn, idx_tst
609        elif split == DatasetSplit.CLHS:
610            n_select = int(round(opt * self.n_samples))
611            if not 1 <= n_select < self.n_samples:
612                raise SpectraXAIValueError("opt leaves an empty training or test set")
613            n_components = max(
614                1, min(clhs_components, self.n_features, self.n_samples - 1)
615            )
616            scores = PCA(
617                n_components=n_components, random_state=random_state
618            ).fit_transform(StandardScaler().fit_transform(self.X))
619            idx_trn = _clhs(
620                scores, n_select, clhs_iterations, np.random.default_rng(random_state)
621            )
622            idx_tst = np.setdiff1d(indices, idx_trn)
623            return idx_trn, idx_tst
624        elif split == DatasetSplit.CROSS_VALIDATION:
625            assert isinstance(opt, int)
626            kf = KFold(opt, shuffle=True, random_state=random_state)
627            idx_trn, idx_tst = [], []
628            for trn_index, tst_index in kf.split(self.X):
629                idx_trn.append(np.array(trn_index))
630                idx_tst.append(np.array(tst_index))
631            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)
632        elif split == DatasetSplit.STRATIFIED:
633            assert isinstance(opt, int)
634            if random_state is None:
635                skf = StratifiedKFold(n_splits=opt)
636            else:
637                skf = StratifiedKFold(
638                    n_splits=opt, shuffle=True, random_state=random_state
639                )
640            idx_trn, idx_tst = [], []
641            for trn_index, tst_index in skf.split(self.X, self._strata(n_bins)):
642                idx_trn.append(np.array(trn_index))
643                idx_tst.append(np.array(tst_index))
644            return self._repair_split_array(idx_trn), self._repair_split_array(idx_tst)
