spectraxai.spectra

The module spectraxai.spectra supports operations on spectra.

This module defines the class Spectra which can be used to hold spectral datasets (only the features, e.g. reflectance values) and perform common operations on them like pre-processing.

 28    """Spectral Preprocessing enum.
 29
 30    A collection of different spectral pre-processing (or pre- treatments) that
 31    may be applied to a spectral matrix.
 32
 33    """
 34
 35    NONE = "no"
 36    REF = "reflectance"
 37    ABS = "absorbance"
 38    SNV = "SNV"
 39    SG0 = "SG0"
 40    SG1 = "SG1"
 41    SG2 = "SG2"
 42    CR = "continuum-removal"
 43    MSC = "MSC"
 44    """Multiplicative scatter correction against a reference (mean) spectrum."""
 45    DETREND = "detrend"
 46    """Remove a linear trend from each spectrum (`scipy.signal.detrend`)."""
 47    MEAN_CENTER = "mean-center"
 48    """Subtract the mean spectrum (column-wise centering)."""

 50    @property
 51    def is_stateful(self) -> bool:
 52        """Whether the transform learns statistics from the (training) samples.
 53
 54        `MSC` and `MEAN_CENTER` depend on the mean spectrum: `Spectra.preprocess`
 55        computes it on the spectra it is given, while
 56        `spectraxai.preprocessing.SpectralPreprocessor` learns it at ``fit`` so
 57        test samples never influence it.
 58
 59        """
 60        return self in (SpectralPreprocessing.MSC, SpectralPreprocessing.MEAN_CENTER)

 62    def __str__(self) -> str:
 63        return self.name
 64
 65    def __repr__(self) -> str:
 66        return self.value

 68    @staticmethod
 69    def _lookup(token: str) -> "SpectralPreprocessing":
 70        """Find a member by its name (e.g. ``ABS``) or value (``absorbance``)."""
 71        for member in SpectralPreprocessing:
 72            if token in (member.name, member.value):
 73                return member
 74        raise SpectraXAIValueError(f"Unknown spectral pre-processing: {token!r}")

 76    @staticmethod
 77    def _from_literal(node: Any) -> Any:
 78        if isinstance(node, str):
 79            return SpectralPreprocessing._lookup(node)
 80        if isinstance(node, list):
 81            return [SpectralPreprocessing._from_literal(item) for item in node]
 82        if isinstance(node, tuple) and len(node) == 2 and isinstance(node[1], dict):
 83            return (SpectralPreprocessing._lookup(node[0]), node[1])
 84        raise SpectraXAIValueError(f"Cannot parse spectral pre-processing: {node!r}")

 86    @staticmethod
 87    def init_class(string: str) -> Any:
 88        """Initialize from its string representation, i.e. the inverse of `str`.
 89
 90        Accepts anything `str` produces for a `SpectralPreprocessingSequence`, such as
 91        the ``pre_process`` column returned by
 92        `spectraxai.models.StandardModel.fit_and_predict`: a single member
 93        (``"ABS"``), a member with options (``"(SG1, {'window_length': 7})"``) or a
 94        (nested) list of those (``"[absorbance, SNV]"``). Members may be written by
 95        name or by value.
 96
 97        Returns
 98        -------
 99        `SpectralPreprocessingSequence`
100            The parsed member, tuple or list (lists are returned as lists).
101
102        """
103        tokens = sorted(
104            {token for m in SpectralPreprocessing for token in (m.name, m.value)},
105            key=len,
106            reverse=True,
107        )
108        # Quote bare tokens only; tokens inside quotes (e.g. dict keys or string
109        # option values) are left untouched.
110        pattern = re.compile(
111            r"""('[^']*'|"[^"]*")|(?<![\w-])("""
112            + "|".join(re.escape(t) for t in tokens)
113            + r""")(?![\w-])"""
114        )
115        quoted = pattern.sub(
116            lambda m: m.group(1) if m.group(1) else repr(m.group(2)), string
117        )
118        try:
119            literal = ast.literal_eval(quoted)
120        except (ValueError, SyntaxError) as error:
121            raise SpectraXAIValueError(
122                f"Cannot parse spectral pre-processing: {string!r}"
123            ) from error
124        return SpectralPreprocessing._from_literal(literal)

146- SpectralPreprocessing.NONE => no pre-treatment

156def _steps(
157    method: SpectralPreprocessingSequence,
158) -> list[tuple[SpectralPreprocessing, dict[str, Any]]]:
159    """Flatten a (nested) pre-processing sequence into ``(method, options)`` steps."""
160    if isinstance(method, SpectralPreprocessing):
161        return [(method, {})]
162    if (
163        isinstance(method, tuple)
164        and len(method) == 2
165        and isinstance(method[0], SpectralPreprocessing)
166        and isinstance(method[1], dict)
167    ):
168        return [(method[0], dict(method[1]))]
169    if isinstance(method, list):
170        return [step for each in method for step in _steps(each)]
171    raise SpectraXAIValueError(f"Unknown spectral pre-processing: {method!r}")

174class Spectra:
175    """Spectra class to hold a 2-D spectral matrix of shape (samples, wavelengths).
176
177    Can accept a 1-D vector as input but always returns a 2-D matrix.
178
179    """
180
181    X: np.ndarray
182    """A 2-D matrix representing the spectra."""
183
184    wavelengths: np.ndarray | None
185    """The wavelength of each column of `X`, if known (used by continuum removal)."""
186
187    def __init__(
188        self, X: np.ndarray, *, wavelengths: np.ndarray | list[float] | None = None
189    ) -> None:
190        """X is a np 2D array containing the (samples, wavelengths) matrix.
191
192        Optionally pass the `wavelengths` of the columns of X.
193
194        """
195        if not _is_numpy_array(X):
196            raise TypeError("X should be a numpy array")
197        if X.ndim == 1:
198            X = X.reshape(1, -1)
199        if X.ndim != 2:
200            raise SpectraXAIValueError("X should be a 2-D matrix")
201        self.X = X
202        self.wavelengths = None
203        if wavelengths is not None:
204            wvs = np.asarray(wavelengths, dtype=float).ravel()
205            if wvs.size != X.shape[1]:
206                raise SpectraXAIValueError(
207                    "wavelengths should have one value per column of X"
208                )
209            self.wavelengths = wvs
210
211    def _new(self, X: np.ndarray) -> "Spectra":
212        return Spectra(X, wavelengths=self.wavelengths)
213
214    def reflectance(self) -> "Spectra":
215        """Transform absorbance to reflectance."""
216        return self._new(1 / 10**self.X)
217
218    def absorbance(self) -> "Spectra":
219        """Transform reflectance to absorbance."""
220        return self._new(-1 * np.log10(self.X))
221
222    def snv(self) -> "Spectra":
223        """Apply the standard normal variate transform."""
224        snv = np.zeros(self.X.shape)
225        mu, sd = self.X.mean(axis=-1), self.X.std(axis=-1, ddof=1)
226        for i in range(np.shape(self.X)[0]):
227            snv[i, :] = (self.X[i, :] - mu[i]) / sd[i]
228        return self._new(snv)
229
230    def sg(self, **kwargs: Any) -> "Spectra":
231        """Apply a general Savitzky–Golay transform.
232
233        You need to pass as kwargs the parameters of
234        `scipy.signal.savgol_filter`
235
236        """
237        return self._new(cast(np.ndarray, savgol_filter(self.X, **kwargs)))
238
239    def cr(self, wavelengths: np.ndarray | list[float] | None = None) -> "Spectra":
240        """Transform absorbance spectra using the Continuum Removal.
241
242        The convex hull is computed against `wavelengths` (or this object's
243        `wavelengths`, or else the column indices). For evenly spaced wavelengths
244        all three give identical results.
245
246        """
247        wvs = self.wavelengths if wavelengths is None else wavelengths
248        wvs_list = None if wvs is None else [float(w) for w in np.ravel(wvs)]
249        return self._new(cast(np.ndarray, continuum_removal(self.X, wvs_list)))
250
251    def msc(self, reference: np.ndarray | None = None) -> "Spectra":
252        """Apply multiplicative scatter correction.
253
254        Each spectrum ``x`` is regressed on the reference ``r`` (``x ≈ a + b r``) and
255        replaced by ``(x - a) / b``.
256
257        Parameters
258        ----------
259        reference: `np.ndarray`, optional
260            The reference spectrum. Defaults to the mean of these spectra.
261
262        """
263        ref = self.X.mean(axis=0) if reference is None else np.ravel(reference)
264        if ref.size != self.X.shape[1]:
265            raise SpectraXAIValueError("reference should have one value per column")
266        design = np.column_stack((np.ones_like(ref), ref))
267        (a, b), *_ = np.linalg.lstsq(design, self.X.T, rcond=None)
268        return self._new((self.X - a[:, None]) / b[:, None])
269
270    def detrend(self, **kwargs: Any) -> "Spectra":
271        """Remove a linear trend from each spectrum.
272
273        Extra kwargs are passed to `scipy.signal.detrend` (e.g. ``bp``).
274
275        """
276        return self._new(cast(np.ndarray, detrend(self.X, axis=-1, **kwargs)))
277
278    def mean_center(self, mean: np.ndarray | None = None) -> "Spectra":
279        """Subtract the mean spectrum (or the given ``mean``) from every spectrum."""
280        mu = self.X.mean(axis=0) if mean is None else np.ravel(mean)
281        if mu.size != self.X.shape[1]:
282            raise SpectraXAIValueError("mean should have one value per column")
283        return self._new(self.X - mu)
284
285    def _apply(
286        self,
287        method: SpectralPreprocessing,
288        options: dict[str, Any],
289        state: np.ndarray | None = None,
290    ) -> "Spectra":
291        """Apply a single transform; ``state`` is the learnt reference, if any."""
292        if method == SpectralPreprocessing.REF:
293            return self.reflectance()
294        elif method == SpectralPreprocessing.ABS:
295            return self.absorbance()
296        elif method == SpectralPreprocessing.SNV:
297            return self.snv()
298        elif method == SpectralPreprocessing.SG0:
299            return self.sg(deriv=0, **options)
300        elif method == SpectralPreprocessing.SG1:
301            return self.sg(deriv=1, **options)
302        elif method == SpectralPreprocessing.SG2:
303            return self.sg(deriv=2, **options)
304        elif method == SpectralPreprocessing.CR:
305            return self.cr(**options)
306        elif method == SpectralPreprocessing.MSC:
307            return self.msc(reference=state)
308        elif method == SpectralPreprocessing.DETREND:
309            return self.detrend(**options)
310        elif method == SpectralPreprocessing.MEAN_CENTER:
311            return self.mean_center(mean=state)
312        elif method == SpectralPreprocessing.NONE:
313            return self
314        raise SpectraXAIValueError(f"Unknown spectral pre-processing: {method!r}")
315
316    def preprocess(self, method: SpectralPreprocessingSequence) -> "Spectra":
317        """Apply a pre-processing sequence (i.e., one or more) specified by method.
318
319        Stateful steps (`SpectralPreprocessing.MSC`,
320        `SpectralPreprocessing.MEAN_CENTER`) use the mean of these spectra. To learn
321        it on training data only, use
322        `spectraxai.preprocessing.SpectralPreprocessor`.
323
324        Parameters
325        ----------
326        method: `spectraxai.spectra.SpectralPreprocessingSequence`
327            The method for the preprocess.
328
329        Returns
330        -------
331        `Spectra`
332            A new Spectra object, where X contains the preprocessed spectra.
333
334        """
335        spectra = self
336        for step, options in _steps(method):
337            spectra = spectra._apply(step, options)
338        return spectra

313        elif method == SpectralPreprocessing.NONE:
