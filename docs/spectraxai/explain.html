spectraxai.explain

The module spectraxai.explain supports the interpretability analysis.

This module defines the class PreHocAnalysis to operate only on a given spectraxai.dataset.Dataset and generate some insights solely from the data, as well as a PostHocAnalysis to operate on a spectraxai.models.StandardModel which has been trained on a given spectraxai.dataset.Dataset.

 114    def _orient_spectral_axis(self, ax: Axes, index_axis: bool) -> None:
 115        """Draw wavenumbers from high to low and wavelengths from low to high.
 116
 117        Idempotent, so it can be applied more than once to the same axes.
 118
 119        """
 120        wavelengths = self.dataset.wavelengths
 121        if index_axis:
 122            if wavelengths is None or wavelengths.size < 2:
 123                return
 124            # Feature indices follow the column order of the data
 125            ascending_data = bool(wavelengths[-1] > wavelengths[0])
 126            inverted = ascending_data == self.dataset.is_wavenumber
 127        else:
 128            inverted = self.dataset.is_wavenumber
 129        if ax.xaxis_inverted() != inverted:
 130            ax.invert_xaxis()

 180    @_styled_unless("ax")
 181    def mean_spectrum(
 182        self,
 183        ax: Axes | None = None,
 184        excluded_bands: Sequence[tuple[float, float]] | None = None,
 185        **kwargs: Any,
 186    ) -> Axes:
 187        r"""Plot the mean spectrum ± standard deviation of the dataset.
 188
 189        Parameters
 190        ----------
 191        ax : Axes, optional
 192            Matplotlib axes to plot into. If None, a new figure is created.
 193
 194        excluded_bands : sequence of tuple[float, float], optional
 195            Inclusive wavelength intervals to leave blank, such as atmospheric
 196            absorption bands.
 197
 198        **kwargs
 199            Passed on to `matplotlib.axes.Axes.plot` for the mean line (e.g.
 200            ``color`` or ``label``); the ±SD band uses the same color.
 201
 202        Returns
 203        -------
 204        Axes
 205            The matplotlib axes with the plot.
 206
 207        """
 208
 209        wavelengths = self._wavelengths()
 210
 211        bands = np.asarray(
 212            excluded_bands if excluded_bands is not None else [],
 213            dtype=float,
 214        ).reshape(-1, 2)
 215
 216        bands = np.sort(bands, axis=1)  # accept (high, low) wavenumber pairs
 217        if not np.isfinite(bands).all():
 218            raise SpectraXAIValueError(
 219                "excluded_bands must contain finite (start, end) wavelength pairs"
 220            )
 221
 222        if ax is None:
 223            _, ax = plt.subplots(figsize=FIGSIZE, dpi=DPI)
 224
 225        order = np.argsort(wavelengths)
 226        wavelengths = wavelengths[order]
 227        spectra = np.asarray(self.dataset.X)[:, order]
 228
 229        mean = spectra.mean(axis=0)
 230        sd = spectra.std(axis=0, ddof=1)
 231
 232        excluded = np.zeros(wavelengths.shape, dtype=bool)
 233        for start, end in bands:
 234            excluded |= (wavelengths >= start) & (wavelengths <= end)
 235        is_valid = ~excluded
 236
 237        if is_valid.any():
 238            # Find indices where validity changes or use split on consecutive indices
 239            valid_indices = np.flatnonzero(is_valid)
 240            # Split into arrays of continuous indices (where index difference > 1)
 241            split_points = np.where(np.diff(valid_indices) > 1)[0] + 1
 242            segments = np.split(valid_indices, split_points)
 243
 244            line_kwargs = {"linewidth": 1.5, **normalize_kwargs(kwargs, Line2D)}
 245            color = line_kwargs.pop("color", None)
 246            label = line_kwargs.pop("label", None)
 247            for seg_idx in segments:
 248                if len(seg_idx) == 0:
 249                    continue
 250
 251                seg_x = wavelengths[seg_idx]
 252                seg_mean = mean[seg_idx]
 253                seg_lower = seg_mean - sd[seg_idx]
 254                seg_upper = seg_mean + sd[seg_idx]
 255
 256                # Plot each continuous segment cleanly
 257                line_plots = ax.plot(
 258                    seg_x, seg_mean, color=color, label=label, **line_kwargs
 259                )
 260                label = None  # one legend entry for all segments
 261                if color is None:
 262                    color = line_plots[0].get_color()
 263
 264                ax.fill_between(
 265                    seg_x,
 266                    seg_lower,
 267                    seg_upper,
 268                    color=color,
 269                    alpha=0.2,
 270                    edgecolor=None,
 271                )
 272        return self._stylize_plot(ax)

 274    @_styled_unless("ax")
 275    def plot_selected_bands(
 276        self,
 277        support: Any,
 278        ax: Axes | None = None,
 279        excluded_bands: Sequence[tuple[float, float]] | None = None,
 280        **kwargs: Any,
 281    ) -> Axes:
 282        """Shade the selected wavelengths over the mean spectrum ± SD.
 283
 284        Parameters
 285        ----------
 286        support: `np.ndarray` or fitted selector
 287            A boolean mask (or integer indices) of the selected features, or a
 288            fitted selector from `spectraxai.selection` (any object with
 289            ``get_support()``).
 290
 291        ax: `plt.Axes`, optional
 292            An optional matplotlib axes to plot into. Defaults to None, in which
 293            case a new figure is created.
 294
 295        excluded_bands: sequence of tuple[float, float], optional
 296            Wavelength intervals to leave blank in the mean spectrum (see
 297            `PreHocAnalysis.mean_spectrum`).
 298
 299        **kwargs
 300            Passed on to `matplotlib.axes.Axes.axvspan` for the shaded bands (e.g.
 301            ``color`` or ``alpha``).
 302
 303        Returns
 304        -------
 305        `plt.Axes`
 306            The matplotlib axes with the plot
 307
 308        """
 309        if hasattr(support, "get_support"):
 310            support = support.get_support()
 311        support = np.asarray(support)
 312        mask = np.zeros(self.dataset.n_features, dtype=bool)
 313        if support.dtype == bool:
 314            if support.shape != (self.dataset.n_features,):
 315                raise SpectraXAIValueError(
 316                    "support should have one value per input feature"
 317                )
 318            mask = support
 319        else:
 320            mask[support.astype(int)] = True
 321        wavelengths = self._wavelengths()
 322        ax = self.mean_spectrum(ax=ax, excluded_bands=excluded_bands)
 323
 324        order = np.argsort(wavelengths)
 325        wvs, selected = wavelengths[order], mask[order]
 326        # Each band spans half-way to its neighbours
 327        mids = (wvs[1:] + wvs[:-1]) / 2
 328        lower = np.concatenate(
 329            ([wvs[0] - (mids[0] - wvs[0]) if mids.size else wvs[0] - 0.5], mids)
 330        )
 331        upper = np.concatenate(
 332            (mids, [wvs[-1] + (wvs[-1] - mids[-1]) if mids.size else wvs[-1] + 0.5])
 333        )
 334        span_kwargs = {
 335            "color": "tab:orange",
 336            "alpha": 0.3,
 337            "linewidth": 0,
 338            "label": "Selected",
 339            **normalize_kwargs(kwargs, Patch),
 340        }
 341        idx = np.flatnonzero(selected)
 342        runs = np.split(idx, np.flatnonzero(np.diff(idx) > 1) + 1) if idx.size else []
 343        for run in runs:
 344            ax.axvspan(float(lower[run[0]]), float(upper[run[-1]]), **span_kwargs)
 345            span_kwargs["label"] = None  # one legend entry
 346        return ax

 349def _normalize_by_max(values: np.ndarray) -> np.ndarray:
 350    """Scale values so that the maximum is 1, leaving all-zero arrays unchanged."""
 351    peak = np.max(values)
 352    return values / peak if peak > 0 else values

 358    def _corr(self) -> np.ndarray:
 359        """Calculate Pearson's correlation between input features and output(s).
 360
 361        Returns
 362        -------
 363        `np.ndarray`
 364            A 2-D np.array containing the correlation for each output property of size
 365            (`spectraxai.dataset.Dataset.n_outputs`,
 366            `spectraxai.dataset.Dataset.n_features`)
 367
 368        """
 369        return np.array(
 370            [
 371                [
 372                    np.corrcoef(self.dataset.X[:, i], self.dataset.Y[:, j])[0][1]
 373                    for i in range(self.dataset.n_features)
 374                ]
 375                for j in range(self.dataset.n_outputs)
 376            ]
 377        )
 378
 379    def _mi(self) -> np.ndarray:
 380        """Calculate the mutual information between all inputs and the output(s).
 381
 382        Returns
 383        -------
 384        `np.ndarray`
 385            A 2-D np.array containing the mutual information for each output property
 386            of size (`spectraxai.dataset.Dataset.n_outputs`,
 387            `spectraxai.dataset.Dataset.n_features`)
 388
 389        """
 390        return np.array(
 391            [
 392                _normalize_by_max(
 393                    mutual_info_regression(
 394                        self.dataset.X,
 395                        self.dataset.Y[:, j],
 396                        random_state=self.random_state,
 397                    )
 398                )
 399                for j in range(self.dataset.n_outputs)
 400            ]
 401        )
 402
 403    def _f_statistic(self) -> np.ndarray:
 404        """Calculate F-statistic in the `dataset`.
 405
 406        Performs univariate linear regression tests returning F-statistic between all
 407        input features and the output(s).
 408
 409        Returns
 410        -------
 411        `np.ndarray`
 412            A 2-D np.array containing the F-statistic for each output property of size
 413            (`spectraxai.dataset.Dataset.n_outputs`,
 414            `spectraxai.dataset.Dataset.n_features`)
 415
 416        """
 417        return np.array(
 418            [
 419                _normalize_by_max(
 420                    np.nan_to_num(f_regression(self.dataset.X, self.dataset.Y[:, j])[0])
 421                )
 422                for j in range(self.dataset.n_outputs)
 423            ]
 424        )

 426    def feature_importance(self, method: object) -> np.ndarray:
 427        """Calculate feature importance between the input features and the output(s).
 428
 429        Parameters
 430        ----------
 431        method: `FeatureRanking`
 432            The method to calculate the feature importance.
 433
 434        Returns
 435        -------
 436        `np.ndarray`
 437            The feature importance according to the selected method, which is a 2-D
 438            np.array containing the ranking for each output property of size
 439            (`spectraxai.dataset.Dataset.n_outputs`,
 440            `spectraxai.dataset.Dataset.n_features`)
 441
 442        """
 443        if not isinstance(method, FeatureRanking):
 444            raise SpectraXAIValueError(
 445                "Method must be one of the types defined in FeatureRanking"
 446            )
 447        if method == FeatureRanking.CORR:
 448            return self._corr()
 449        elif method == FeatureRanking.MI:
 450            return self._mi()
 451        elif method == FeatureRanking.F_STATISTIC:
 452            return self._f_statistic()
 453        raise SpectraXAIValueError(f"Unsupported feature ranking: {method!r}")
 454
 455    @_styled_unless("axes")
 456    def correlogram(
 457        self,
 458        top: int = 5,
 459        method: FeatureRanking = FeatureRanking.CORR,
 460        *,
 461        axes: np.ndarray | Axes | None = None,
 462        **kwargs: Any,
 463    ) -> np.ndarray:
 464        """Plot a correlogram between the most important inputs and the output(s).
 465
 466        Parameters
 467        ----------
 468        top: `int`, optional
 469            The number of most important features to consider. Defaults to 5.
 470
 471        method: `FeatureRanking`, optional
 472            The method to calculate the feature importance.
 473            Defaults to FeatureRanking.CORR.
 474
 475        axes: array of `plt.Axes`, optional
 476            ``n_outputs * top`` axes to draw into (any shape; read row by row, one
 477            row per output). Defaults to None, in which case a new figure is created.
 478
 479        **kwargs
 480            Passed on to `matplotlib.axes.Axes.scatter`.
 481
 482        Returns
 483        -------
 484        `np.ndarray`
 485            A 2-D array of shape (`n_outputs`, `top`) with the matplotlib axes
 486
 487        """
 488        if not 1 <= top <= self.dataset.n_features:
 489            raise SpectraXAIValueError(
 490                "top should be between 1 and the number of input features"
 491            )
 492        metric = self.feature_importance(method)
 493        created = axes is None
 494        if axes is None:
 495            _, axes = plt.subplots(
 496                self.dataset.n_outputs, top, figsize=FIGSIZE, squeeze=False, dpi=DPI
 497            )
 498        else:
 499            axes = _as_axes_array(
 500                axes, self.dataset.n_outputs * top, "n_outputs * top"
 501            ).reshape(self.dataset.n_outputs, top)
 502        scatter_kwargs = {"s": 4, **kwargs}
 503        for i, corr in enumerate(metric):
 504            ind = np.argpartition(np.abs(corr), -top)[-top:]
 505            ind = ind[np.argsort(corr[ind])]
 506            for j in range(top):
 507                x, y = self.dataset.X[:, ind[j]], self.dataset.Y[:, i]
 508                axes[i, j].scatter(x=x, y=y, **scatter_kwargs)
 509                y_lim = axes[i, j].get_ylim()
 510                m, b = np.polyfit(x, y, 1)
 511                axes[i, j].plot(x, m * x + b, c="k")
 512                axes[i, j].set_ylim(y_lim)
 513                axes[i, j].set_title("{} {:.2f}".format(method.value, corr[ind[j]]))
 514                axes[i, j].set_xlabel("Feature {}".format(self.dataset.X_names[ind[j]]))
 515                axes[i, j].set_ylabel("Output {}".format(self.dataset.Y_names[i]))
 516        if created:
 517            axes[0, 0].get_figure().tight_layout()
 518        return axes

 634            excluded_bands if excluded_bands is not None else [],
 635            dtype=float,
